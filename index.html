from playwright.sync_api import Page, expect
from Log_in_page import Loginpage

def test_sucesful_login(page: Page):
    """Test successful user login & inventory redirection."""
    page.goto("https://www.saucedemo.com/")
    login_page = Loginpage(page)

    # Perform login and assert target inventory route
    login_page.login("standard_user", "secret_sauce")
    expect(page).to_have_url("https://www.saucedemo.com/inventory.html")

def test_unsucesful_login(page: Page):
    """Assert error banner appearance on invalid credentials."""
    page.goto("https://www.saucedemo.com/")
    login_page = Loginpage(page)
    login_page.login("nobody", "123")
    
    # Assertion via encapsulated POM locator
    expect(login_page.login_error()).to_be_visible()
from playwright.sync_api import Page

class Loginpage:
    """Page Object Model encapsulating the SauceDemo authentication page."""
    def __init__(self, page: Page):
        self.page = page
        self.username_input = page.locator("#user-name")
        self.password_input = page.locator("#password")
        self.loginbutton = page.locator("#login-button")
        self.loginerrorbttn = page.locator("[data-test='error']")

    def login(self, jmeno: str, heslo: str):
        # Reusable atomic action
        self.username_input.fill(jmeno)
        self.password_input.fill(heslo)
        self.loginbutton.click()

    def login_error(self):
        return self.loginerrorbttn
